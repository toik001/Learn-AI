Matt Pocock Skills 安装与 Java 练习环境

总览 · 安装与环境 · 完整实战 · 原理说明

第 1 课：准备一个独立的 Java 练习仓库，安装固定版本技能，跑通基础订单系统。预计 45～90 分钟，首次依赖下载另计。

资料核验：2026-09-16；skills 固定到 959a8e9f1edc3adbe2f7e3054bb6fbefa6696260。教程代码为本课程编写，不是 Matt 官方 Java 示例。

核心结论

先证明工程能运行，再用技能增加功能。基础版已经有登录、订单查询、模拟支付和数据库；第二课只增加取消订单，便于观察每一步如何改变产物。

源代码保存在本页与下一页的 HTML 代码块里。提取器读取代码块，在知识库外生成普通 Maven 项目；本知识库不保存第二份 Java 源码或 Markdown 学习笔记。所有练习命令都在生成的订单项目执行，只有提取命令在 Learn-AI 根目录执行。

package study.orders;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpStatus;
import org.springframework.security.config.Customizer;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.core.userdetails.User;
import org.springframework.security.core.userdetails.UserDetailsService;
import org.springframework.security.crypto.factory.PasswordEncoderFactories;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.provisioning.InMemoryUserDetailsManager;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.HttpStatusEntryPoint;
import org.springframework.security.web.util.matcher.RequestMatcher;

@Configuration
public class SecurityConfig {
    @Bean PasswordEncoder passwords() {
        return PasswordEncoderFactories.createDelegatingPasswordEncoder();
    }

    @Bean UserDetailsService users(PasswordEncoder encoder,
            @Value("${demo.alice-password}") String alice,
            @Value("${demo.bob-password}") String bob) {
        return new InMemoryUserDetailsManager(
            User.withUsername("alice").password(encoder.encode(alice)).roles("USER").build(),
            User.withUsername("bob").password(encoder.encode(bob)).roles("USER").build());
    }

    @Bean SecurityFilterChain web(HttpSecurity http) throws Exception {
        RequestMatcher api = request -> request.getServletPath().startsWith("/api/");
        return http
            .authorizeHttpRequests(auth -> auth.requestMatchers("/api/csrf").permitAll()
                .anyRequest().authenticated())
            .formLogin(Customizer.withDefaults())
            .exceptionHandling(errors -> errors.defaultAuthenticationEntryPointFor(
                new HttpStatusEntryPoint(HttpStatus.UNAUTHORIZED), api))
            .build(); // Default CSRF protection remains enabled.
    }
}

package study.orders;

import java.util.List;
import org.springframework.http.HttpStatus;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.jdbc.core.RowMapper;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;

@Service
public class OrderService {
    private final JdbcTemplate jdbc;
    private static final RowMapper<OrderView> ROW = (rs, row) -> new OrderView(
        rs.getLong("id"), rs.getString("owner"), rs.getString("status"), rs.getLong("version"));

    public OrderService(JdbcTemplate jdbc) { this.jdbc = jdbc; }

    public List<OrderView> list(String owner) {
        return jdbc.query("SELECT * FROM customer_order WHERE owner = ? ORDER BY id", ROW, owner);
    }

    public OrderView get(long id, String owner) { return find(id, owner, false); }

    @Transactional
    public OrderView pay(long id, String owner) { return transition(id, owner, "PAID"); }

    private OrderView transition(long id, String owner, String target) {
        OrderView current = find(id, owner, true);
        if (current.status().equals(target)) return current;
        if (!current.status().equals("UNPAID")) {
            throw new ResponseStatusException(HttpStatus.CONFLICT, "STATE_CONFLICT");
        }
        jdbc.update("UPDATE customer_order SET status = ?, version = version + 1 WHERE id = ? AND owner = ?",
            target, id, owner);
        return find(id, owner, false);
    }

    private OrderView find(long id, String owner, boolean lock) {
        List<OrderView> found = jdbc.query(
            "SELECT * FROM customer_order WHERE id = ? AND owner = ?" + (lock ? " FOR UPDATE" : ""),
            ROW, id, owner);
        if (found.isEmpty()) throw new ResponseStatusException(HttpStatus.NOT_FOUND, "ORDER_NOT_FOUND");
        return found.getFirst();
    }
}

package study.orders;

import java.security.Principal;
import java.util.List;
import java.util.Map;
import org.springframework.security.web.csrf.CsrfToken;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RestController;

@RestController
public class OrderController {
    private final OrderService orders;
    public OrderController(OrderService orders) { this.orders = orders; }

    @GetMapping("/api/csrf")
    Map<String, String> csrf(CsrfToken token) {
        return Map.of("headerName", token.getHeaderName(), "token", token.getToken());
    }

    @GetMapping("/api/orders")
    List<OrderView> list(Principal user) { return orders.list(user.getName()); }

    @GetMapping("/api/orders/{id}")
    OrderView get(@PathVariable long id, Principal user) { return orders.get(id, user.getName()); }

    @PostMapping("/api/orders/{id}/pay")
    OrderView pay(@PathVariable long id, Principal user) { return orders.pay(id, user.getName()); }
}

const message = document.getElementById('message');
let csrf;
async function request(url, method = 'GET') {
  const response = await fetch(url, {
    method,
    headers: method === 'GET' ? {} : { [csrf.headerName]: csrf.token },
    credentials: 'same-origin'
  });
  if (response.redirected || response.status === 401) {
    location.assign('/login');
    throw new Error('请重新登录');
  }
  if (!response.ok) throw new Error(`请求失败：HTTP ${response.status}`);
  return response.status === 204 ? null : response.json();
}
async function loadOrders() {
  const orders = await request('/api/orders');
  const body = document.getElementById('orders');
  body.replaceChildren();
  for (const order of orders) {
    const row = document.createElement('tr');
    for (const value of [order.id, order.owner, order.status, order.version]) {
      const cell = document.createElement('td');
      cell.textContent = String(value);
      row.append(cell);
    }
    const actions = document.createElement('td');
    addAction(actions, order, 'pay', '模拟支付', order.status === 'UNPAID');
    row.append(actions);
    body.append(row);
  }
}
function addAction(cell, order, action, label, enabled) {
  const button = document.createElement('button');
  button.textContent = label;
  button.disabled = !enabled;
  button.addEventListener('click', async () => {
    button.disabled = true;
    try {
      const result = await request(`/api/orders/${order.id}/${action}`, 'POST');
      message.textContent = `订单 ${result.id}：${result.status}，版本 ${result.version}`;
    } catch (error) { message.textContent = error.message; }
    finally { await loadOrders().catch(error => { message.textContent = error.message; }); }
  });
  cell.append(button);
}
document.getElementById('refresh').addEventListener('click', () => {
  loadOrders().catch(error => { message.textContent = error.message; });
});
document.getElementById('logout').addEventListener('click', async () => {

    @BeforeEach void prepare() throws Exception {
        jdbc.update("DELETE FROM customer_order");
        jdbc.update("INSERT INTO customer_order VALUES (101, 'alice', 'UNPAID', 0), (102, 'alice', 'PAID', 0), (103, 'bob', 'UNPAID', 0)");
        alice = actor("alice", "alice-test-only");
        bob = actor("bob", "bob-test-only");
    }

    URI url(String path) { return URI.create("http://127.0.0.1:" + port + path); }

    Actor actor(String user, String password) throws Exception {
        HttpClient client = HttpClient.newBuilder().connectTimeout(Duration.ofSeconds(5))
            .cookieHandler(new CookieManager(null, CookiePolicy.ACCEPT_ALL)).build();
        var beforeLogin = csrf(client);
        String header = json(beforeLogin, "$.headerName");
        String token = json(beforeLogin, "$.token");
        String form = "username=" + URLEncoder.encode(user, StandardCharsets.UTF_8)
            + "&password=" + URLEncoder.encode(password, StandardCharsets.UTF_8);
        var login = client.send(HttpRequest.newBuilder(url("/login"))
            .header("Content-Type", "application/x-www-form-urlencoded").header(header, token)
            .timeout(Duration.ofSeconds(10)).POST(HttpRequest.BodyPublishers.ofString(form)).build(),
            HttpResponse.BodyHandlers.ofString());
        if (login.statusCode() != 302 || login.headers().firstValue("Location").orElse("").contains("error")) {
            throw new IllegalStateException("Login failed: " + login.statusCode());
        }
        // Authentication rotates the CSRF token, so fetch a fresh token with the session cookie.
        var afterLogin = csrf(client);
        return new Actor(client, json(afterLogin, "$.headerName"), json(afterLogin, "$.token"));
    }

    private HttpResponse<String> csrf(HttpClient client) throws Exception {
        var response = client.send(HttpRequest.newBuilder(url("/api/csrf"))
            .timeout(Duration.ofSeconds(10)).build(), HttpResponse.BodyHandlers.ofString());
        if (response.statusCode() != 200) throw new IllegalStateException("CSRF setup failed: " + response.statusCode());
        return response;
    }

    HttpResponse<String> send(Actor actor, String method, String path) throws Exception {
        return send(actor, method, path, true);
    }

    HttpResponse<String> send(Actor actor, String method, String path, boolean token) throws Exception {
        var request = HttpRequest.newBuilder(url(path)).timeout(Duration.ofSeconds(10))
            .header("Accept", "application/json").method(method, HttpRequest.BodyPublishers.noBody());
        if (token && !method.equals("GET")) request.header(actor.csrfHeader(), actor.csrfToken());
        return actor.client().send(request.build(), HttpResponse.BodyHandlers.ofString());
    }

    <T> T json(HttpResponse<String> response, String expression) {
        return JsonPath.read(response.body(), expression);
    }

    String statusOf(Actor actor, long id) throws Exception {
        return json(send(actor, "GET", "/api/orders/" + id), "$.status");
    }

    int versionOf(Actor actor, long id) throws Exception {
        Number value = json(send(actor, "GET", "/api/orders/" + id), "$.version");
        return value.intValue();
    }
}

7. 常见问题

现象定位与恢复
输出目录非空换一个空目录；不要为重试删除已有练习成果。
Java 版本不对检查 mvn -version 实际使用的 JDK；pom.xml 要求 Java 25。
依赖下载失败检查 Maven 网络/镜像；先跑基础测试，避免把网络失败当成业务测试的红灯。
启动提示缺少密码在同一终端导出 DEMO_ALICE_PASSWORD、DEMO_BOB_PASSWORD。
8080 被占用使用 PORT=8081 mvn spring-boot:run，并访问相应端口。
POST 返回 403先检查登录态与 CSRF token；不要通过禁用 CSRF 修复测试。
演示订单已改变使用新的练习目录，或显式备份并更换开发数据库路径；不要删除来源不明的数据。
技能无法找到其他技能检查 grilling、domain-modeling、codebase-design 和 agents/openai.yaml 是否一起复制。

镜像故障恢复：隔离全局与用户 Maven 设置

本次验证确实遇到本机全局镜像下载超时。若你也遇到依赖下载失败，可在独立练习目录创建以下临时配置，显式直连 Maven Central；不要改动全局配置。-s 只替代用户配置，仍可能受到全局镜像影响，因此这里同时指定 -gs。单位网络若要求使用其镜像，应按本单位网络规则修复镜像。
